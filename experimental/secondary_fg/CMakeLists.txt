cmake_minimum_required(VERSION 3.21)
project(mgpu_secondary_fg_prototype LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()

add_executable(frame_contract_test frame_contract_test.cpp)
add_test(NAME frame_contract_test COMMAND frame_contract_test)
add_executable(camera_contract_test camera_contract_test.cpp)
add_test(NAME camera_contract_test COMMAND camera_contract_test)

if(WIN32 AND MSVC)
    add_library(dlss_fg_stage STATIC dlss_fg_stage.cpp parameter_bridge.cpp)
    target_compile_definitions(dlss_fg_stage PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    add_executable(stage_api_test stage_api_test.cpp parameter_bridge_test.cpp)
    target_compile_definitions(stage_api_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(stage_api_test PRIVATE dlss_fg_stage dxguid)
    add_test(NAME stage_api_test COMMAND stage_api_test)
    if(MSVC)
        add_executable(fg_inference_probe fg_inference_probe.cpp parameter_bridge.cpp)
    target_compile_definitions(fg_inference_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(fg_inference_probe PRIVATE dlss_fg_stage dxguid dxgi d3d12)
    add_executable(capability_probe capability_probe.cpp parameter_bridge.cpp)
        target_compile_definitions(capability_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
        target_link_libraries(capability_probe PRIVATE dxgi dxguid d3d12)
    else()
        message(STATUS "Skipping CMake probe: use build_probe.ps1 for its isolated MSVC-ABI parameter bridge.")
    endif()
endif()
